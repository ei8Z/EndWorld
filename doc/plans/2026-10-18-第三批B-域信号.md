```kotlin
package com.ei8z.endworld.domain.npc.engine

import com.ei8z.endworld.domain.npc.model.NpcState
import com.ei8z.endworld.domain.world.engine.WorldRulesEngine.BattleResult
import kotlin.random.Random

enum class CasualtySeverity { WOUNDED, SEVERE }

data class Casualty(
    val npcId: String,
    val severity: CasualtySeverity,
    val battleId: String,
    val factionId: String
)

/**
 * 战斗伤亡抽取：只在参战势力内、按该侧伤亡率 × 角色修正抽签。
 * 中签只代表"受伤/重伤"，是否死亡由场景3 的裁决结果决定。
 */
class BattleCasualtyEngine(private val random: Random = Random.Default) {

    fun roll(npcs: Collection<NpcState>, battle: BattleResult): List<Casualty> {
        val out = mutableListOf<Casualty>()
        for (npc in npcs) {
            val factionId = npc.factionId ?: continue
            val rate = when (factionId) {
                battle.attackerFactionId -> battle.attackerCasualtyRate
                battle.defenderFactionId -> battle.defenderCasualtyRate
                else -> continue
            }
            val probability = (rate * npc.combatRole.deathProbabilityModifier).coerceIn(0f, 1f)
            if (random.nextFloat() >= probability) continue
            val severity = if (random.nextFloat() < 0.3f) CasualtySeverity.SEVERE else CasualtySeverity.WOUNDED
            out.add(Casualty(npc.npcId, severity, "${battle.attackerFactionId}-${battle.defenderFactionId}", factionId))
        }
        return out
    }
}
```

```kotlin
interface MemosPauseStore {
    fun failures(npcId: String): Int
    fun setFailures(npcId: String, value: Int)
    fun pausedUntil(npcId: String): Int?
    fun setPausedUntil(npcId: String, day: Int?)
}

class InMemoryMemosPauseStore : MemosPauseStore {
    private val failures = mutableMapOf<String, Int>()
    private val paused = mutableMapOf<String, Int>()
    override fun failures(npcId: String) = failures[npcId] ?: 0
    override fun setFailures(npcId: String, value: Int) { failures[npcId] = value }
    override fun pausedUntil(npcId: String) = paused[npcId]
    override fun setPausedUntil(npcId: String, day: Int?) {
        if (day == null) paused.remove(npcId) else paused[npcId] = day
    }
}

class PrefsMemosPauseStore(private val prefs: SharedPreferences) : MemosPauseStore {
    override fun failures(npcId: String) = prefs.getInt("memos_fail_$npcId", 0)
    override fun setFailures(npcId: String, value: Int) {
        prefs.edit().putInt("memos_fail_$npcId", value).apply()
    }
    override fun pausedUntil(npcId: String): Int? =
        prefs.getInt("memos_pause_$npcId", -1).takeIf { it >= 0 }
    override fun setPausedUntil(npcId: String, day: Int?) {
        if (day == null) prefs.edit().remove("memos_pause_$npcId").apply()
        else prefs.edit().putInt("memos_pause_$npcId", day).apply()
    }
```
