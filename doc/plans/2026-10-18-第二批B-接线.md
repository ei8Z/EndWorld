`ReasonerApiClient`：

```kotlin
    companion object {
        /** 工具调用最多两轮，避免模型反复查询 */
        private const val MAX_TOOL_ROUNDS = 2
    }

    suspend fun reason(
        systemPrompt: String,
        userMessage: String,
        userId: String,
        tools: List<Tool>? = null
    ): ActionResult {
        var messages = listOf(
            ChatMessage(role = "system", content = systemPrompt),
            ChatMessage(role = "user", content = userMessage)
        )

        return try {
            repeat(MAX_TOOL_ROUNDS + 1) { round ->
                val request = ChatRequest(
                    model = "deepseek-flash",
                    messages = messages,
                    reasoningEffort = "max",
                    responseFormat = ResponseFormat("json_object"),
                    maxTokens = 8192,
                    stream = false,
                    userId = userId,
                    tools = tools,
                    toolChoice = if (tools.isNullOrEmpty()) "none" else "auto"
                )
                val response = apiService.getChatCompletions(authHeader, request)
                val choice = response.choices.firstOrNull()
                val toolCalls = choice?.message?.toolCalls

                if (!toolCalls.isNullOrEmpty() && round < MAX_TOOL_ROUNDS) {
                    val toolMessages = toolCalls.map { call ->
                        val output = ToolRegistry.execute(call.function.name, call.function.arguments)
                            ?: """{"error":"unknown tool: ${call.function.name}"}"""
                        ChatMessage(role = "tool", content = output, toolCallId = call.id)
                    }
                    messages = messages +
                        ChatMessage(role = "assistant", content = null, toolCalls = toolCalls) +
                        toolMessages
                    return@repeat
                }

                val content = choice?.message?.content
                return if (content.isNullOrBlank()) {
                    buildFallbackResult(userMessage)
                } else {
                    parseActionResult(content)
                }
            }
            buildFallbackResult(userMessage)
        } catch (e: CancellationException) {
            throw e
        } catch (e: Exception) {
            buildFallbackResult(userMessage)
        }
    }
```

```kotlin
package com.ei8z.endworld.data.repository

import com.ei8z.endworld.domain.npc.model.toRelationshipStage

/** 轻量 AI 场景编号，对应 Agent 配置规范 §5.5-5.10。 */
enum class SceneType(val id: Int) {
    STAGE_CROSSING(2),
    KEY_NPC_DEATH(3),
    NEEDS_MUTATION(4),
    RARE_INTERACTION(5),
    TIMEOUT(6),
    SCHEDULE_ANOMALY(7)
}

data class SceneTrigger(
    val type: SceneType,
    val npcId: String,
    val detail: Map<String, String> = emptyMap()
)

object SceneTriggerDetector {

    /** 好感度跨越关系阶段边界（-60/-20/+20/+50/+80）。 */
    fun detectStageCrossing(npcId: String, oldAffinity: Int, newAffinity: Int): SceneTrigger? {
        val oldStage = oldAffinity.toRelationshipStage()
        val newStage = newAffinity.toRelationshipStage()
        if (oldStage == newStage) return null
        return SceneTrigger(
            type = SceneType.STAGE_CROSSING,
            npcId = npcId,
            detail = mapOf(
                "oldAffinity" to oldAffinity.toString(),
                "newAffinity" to newAffinity.toString(),
                "oldStage" to oldStage.name,
                "newStage" to newStage.name
            )
        )
    }
}
```

```kotlin
package com.ei8z.endworld.data.repository

import android.util.Log
import com.google.gson.JsonObject
import com.google.gson.JsonParser

/**
 * 场景 2-7 的统一评估入口：
 * 提示词 → 裁决客户端（非流式）→ 原始 JSON → LightweightAIValidator 白名单校验。
 * 校验失败返回 null，调用方按 §5.12 降级策略处理。
 */
class SceneEvaluator(private val reasoner: ReasonerApiClient) {

    suspend fun evaluate(trigger: SceneTrigger, userMessage: String): JsonObject? {
        val raw = reasoner.completeRaw(
            systemPrompt = PromptBuilder.buildLightweightEvalSystemPrompt(),
            userMessage = userMessage,
            userId = "scene_${trigger.type.id}"
        ) ?: return null
        val json = try {
            JsonParser.parseString(raw).asJsonObject
        } catch (e: Exception) {
            Log.w("SceneEvaluator", "场景 ${trigger.type.id} 返回非 JSON")
            return null
        }
        val result = LightweightAIValidator.validate(json, trigger.type.id)
        if (!result.isValid) {
            Log.w("SceneEvaluator", "场景 ${trigger.type.id} 校验失败: ${result.warnings}")
            return null
        }
        return result.correctedJson
    }
}
```

```kotlin
    fun detectNeedsMutation(npcId: String, previous: NeedState, current: NeedState): SceneTrigger? {
        val saturatedNow = listOf(

            type = SceneType.NEEDS_MUTATION,
            npcId = npcId,
            detail = mapOf("mutation" to saturatedNow.joinToString("、") { it.first })
        )
    }
```

```kotlin
class MemosExtractionManager(
    private val memosApiService: MemosApiService,
    private val memosApiKey: String
) {
    private val consecutiveFailures = mutableMapOf<String, Int>()
    private val pausedUntilDay = mutableMapOf<String, Int>()

    fun isExtractionPaused(npcId: String, gameDay: Int): Boolean {
        val until = pausedUntilDay[npcId] ?: return false
        return gameDay < until
    }

    suspend fun extractMemories(
        npcId: String,
        interactionTexts: List<String>,
        timestamp: String,
        location: String,
        gameDay: Int
    ): List<MemoryEntry> {
        if (isExtractionPaused(npcId, gameDay)) return emptyList()
        if (interactionTexts.isEmpty()) return emptyList()

        return try {
            val request = MemosExtractRequest(
                messages = interactionTexts.mapIndexed { i, text ->
                    MemosMessage(role = if (i % 2 == 0) "user" else "assistant", content = text)
                },
                extractionTypes = listOf("memory", "preference")
            )
            val response = memosApiService.extractMemory("Token $memosApiKey", request)
            if (response.code != 0) {
                recordFailure(npcId, gameDay)
                return emptyList()
            }
            val data = response.data ?: run {
                recordFailure(npcId, gameDay)
                return emptyList()
            }
            consecutiveFailures.remove(npcId)
            pausedUntilDay.remove(npcId)
            data.memories.orEmpty().map { m ->
                MemoryEntry(
                    timestamp = timestamp,
                    location = location,
                    summary = "[${m.key ?: ""}]: ${m.value ?: ""}",
                    source = MemorySource.MEMOS_EXTRACTED,
                    emotionalTag = m.tags?.firstOrNull()
                )
            }
        } catch (e: kotlinx.coroutines.CancellationException) {
            throw e
        } catch (e: Exception) {
            recordFailure(npcId, gameDay)
            emptyList()
        }
    }

    private fun recordFailure(npcId: String, gameDay: Int) {
        val failures = (consecutiveFailures[npcId] ?: 0) + 1
        consecutiveFailures[npcId] = failures
        if (failures >= 5) {
            pausedUntilDay[npcId] = gameDay + 30
        }
    }
}
```
